const createDiaryByDate = async (dateStr) => {
    const folderPath = "Diary";
    const filePath = `${folderPath}/${dateStr}.md`;
    const file = app.vault.getAbstractFileByPath(filePath);
    
    // 如果文件已存在，直接打开
    if (file) {
        app.workspace.getLeaf(false).openFile(file); 
        return;
    }
    
    let content = "";
    const isToday = dateStr === window.moment().format('YYYY-MM-DD');
    
    // 区分是否为当天，决定是否使用模板
    if (isToday) {
        try {
            const templateFile = app.vault.getAbstractFileByPath("Templates/new_diary.md");
            if (templateFile) {
                content = await app.vault.read(templateFile);
                content = content.replace(/<% tp\.date\.now\("YYYY-MM-DD"\) %>/g, dateStr);
            } else {
                throw new Error("未找到模板文件");
            }
        } catch (e) {
            new Notice(`⚠️ ${e.message}，已使用默认格式`);
            content = getDefaultContent(dateStr);
        }
    } else {
        // 非今日使用固定格式
        content = getDefaultContent(dateStr);
    }

    try {
        const newFile = await app.vault.create(filePath, content);
        
        // 创建成功后，直接打开该笔记
        app.workspace.getLeaf(false).openFile(newFile);
        new Notice(`✅ 已创建并打开 ${dateStr} 的日记`);
        
    } catch (err) {
        new Notice("❌ 创建失败，请检查路径或权限");
    }
};

const parseDateStr = (d) => {
    if (!d) return "";
    if (d.toFormat) return d.toFormat("yyyy-MM-dd");
    return window.moment(d.toString().substring(0, 10)).format('YYYY-MM-DD');
};

const allSortedPages = dv.pages('"Diary"').where(p => p.date && p.file.name !== "new_diary").sort(p => p.date, 'desc');
const tabContainer = dv.el("div", "");
tabContainer.className = "timeline-tab-container";
const timelineContainer = dv.el("div", "");

const getFilteredPages = (type) => {
    const now = window.moment();
    return allSortedPages.filter(p => {
        const pDate = window.moment(parseDateStr(p.date));
        switch(type) {
            case 'recent': return pDate.isSameOrAfter(now.clone().subtract(7, 'days'));
            case 'week': return pDate.isSameOrAfter(now.clone().startOf('isoWeek'));
            case 'month': return pDate.isSameOrAfter(now.clone().startOf('month'));
            case 'year': return pDate.isSameOrAfter(now.clone().startOf('year'));
            default: return true;
        }
    });
};

const renderTimeline = (type = 'all') => {
    const filtered = getFilteredPages(type);
    
    tabContainer.querySelectorAll('.timeline-tab-btn').forEach(btn => {
        btn.classList.remove('active');
        if(btn.getAttribute('data-type') === type) btn.classList.add('active');
    });

    if (filtered.length === 0) {
        timelineContainer.innerHTML = `<div class="timeline-empty">该时间段内还没有日记记录哦~</div>`;
        return;
    }

    let timelineHTML = `<div class="diary-timeline">`;
    filtered.forEach((p) => {
        let preview = "点击卡片查看详情...";
        if (p.review) {
            const match = p.review.match(/([^。！？\n]+)/);
            if (match) preview = match[0] + "...";
        }
        let tagsHTML = p.tags ? p.tags.map(t => `<span class="mini-tag">${t}</span>`).join("") : "";
        const dateStr = parseDateStr(p.date);
        const mtimeStr = window.moment(p.file.mtime).format('YYYY-MM-DD HH:mm');
        const wordCount = Math.round((p.file.size || 0) / 2);
// 【修改点】拆分天气和心情，并赋予独立的胶囊标签类名
        let metaHTML = "";
        if (p.weather) metaHTML += `<span class="timeline-meta meta-weather">${p.weather}</span>`;
        if (p.mood) metaHTML += `<span class="timeline-meta meta-mood">${p.mood}</span>`;
        timelineHTML += `
        <div class="timeline-item">
            <div class="timeline-dot"></div>
            <div class="timeline-content">
                <div class="timeline-date">${dateStr}</div>
                <a class="internal-link timeline-card-link" href="${p.file.path}">
                    <div class="timeline-header">
                        <span class="timeline-title">${p.file.name}</span>
                        <div class="timeline-meta-group">
                            ${metaHTML}
                        </div>
                    </div>

	                    <div>
		                    <div class="timeline-tags">${tagsHTML}</span></div>

	                    </div>
                        <span>🕒 最后编辑: ${mtimeStr}</span>
                    </div>
                </a>
            </div>
        </div>`;
    });
    timelineHTML += `</div>`;
    timelineContainer.innerHTML = timelineHTML;
};
